cmake_minimum_required(VERSION 3.20)
project(nv2a_texture_signature C)
set(CMAKE_C_STANDARD 11)
add_executable(nv2a_texture_signature_test test_main.c)
target_include_directories(nv2a_texture_signature_test PRIVATE ../../src)
if(MSVC)
    target_compile_options(nv2a_texture_signature_test PRIVATE /UNDEBUG)
else()
    target_compile_options(nv2a_texture_signature_test PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME nv2a_texture_signature COMMAND nv2a_texture_signature_test)
